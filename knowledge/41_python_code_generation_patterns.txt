PATTERN 2: VALIDATE INPUT

```python
def is_valid_choice(choice):
    """Return True when choice is one of the allowed values."""
    allowed = ["rock", "paper", "scissors"]
    return choice.lower() in allowed
```

```python
def classify_number(number):
    if number > 0:
        return "positive"
    elif number < 0:
        return "negative"
    return "zero"
```

PATTERN 4: LOOP THROUGH A LIST

```python
def print_items(items):
```

PATTERN 5: COUNT WORDS WITH A DICTIONARY

```python
def count_words(sentence):
    counts = {}
    words = sentence.lower().split()

    for word in words:
        if word not in counts:
            counts[word] = 0
        counts[word] += 1

    return counts
```

PATTERN 6: RANDOM CHOICE

PATTERN 7: SAFE NUMBER INPUT

```python
def get_number():
    while True:
        try:
            return float(input("Enter a number: "))
        except ValueError:
            print("Please enter a valid number.")
```

PATTERN 8: READ A TEXT FILE

```python
def read_text_file(filename):
    with open(filename, "r", encoding="utf-8") as file:
        return file.read()
```

PATTERN 9: WRITE JSON

```python
import json


def save_settings(settings, filename):
    with open(filename, "w", encoding="utf-8") as file:
        json.dump(settings, file, indent=4)
```
